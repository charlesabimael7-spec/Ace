RunService.RenderStepped:Connect(function()
	local ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
	local fps  = math.floor(1 / RunService.RenderStepped:Wait() * 0 + 0)  -- placeholder, see below
	WM_PingFps.Text = tostring(ping) .. " | " .. tostring(math.floor(1/0.016))
end)
local lastTick = tick()
local fpsCount = 0
local currentFPS = 60
RunService.RenderStepped:Connect(function()
	fpsCount = fpsCount + 1
	if tick() - lastTick >= 0.5 then
		currentFPS = math.floor(fpsCount / (tick() - lastTick))
		fpsCount = 0
		lastTick = tick()
	end
	local ok, ping = pcall(function()
		return math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
	end)
	WM_PingFps.Text = (ok and tostring(ping) or "?") .. " | " .. tostring(currentFPS)
	-- fill bar mit FPS ratio
	local ratio = math.clamp(currentFPS / 60, 0, 1)
	TweenService:Create(WM_BarFill, TweenInfo.new(0.3), {Size = UDim2.new(ratio, 0, 1, 0)}):Play()
end)

local function createToggle(parent, labelText, yPos, stateKey)
	-- Label
	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(0, 108, 0, 15)
	lbl.Position = UDim2.new(0.08, 0, 0, yPos)
	lbl.BackgroundTransparency = 1
	lbl.Text = labelText
	lbl.TextColor3 = COLOR_TEXT
	lbl.Font = Enum.Font.Gotham
	lbl.TextSize = 9
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = parent
	-- Toggle Button (der "Schalter")
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(0, 28, 0, 13)
	btn.Position = UDim2.new(0.92, -30, 0, yPos + 1)
	btn.BackgroundColor3 = toggleStates[stateKey] and COLOR_BLUE or Color3.new(0.15, 0.17, 0.25)
	btn.BorderSizePixel = 0
	btn.Text = ""
	btn.AutoButtonColor = false
	btn.Parent = parent
	local btn_Corner = Instance.new("UICorner")
	btn_Corner.CornerRadius = UDim.new(1, 0)
	btn_Corner.Parent = btn
	local btn_Stroke = Instance.new("UIStroke")
	btn_Stroke.Thickness = 1.2
	btn_Stroke.Transparency = 0.5
	btn_Stroke.Parent = btn
	-- Knob
	local knob = Instance.new("Frame")
	knob.Size = UDim2.new(0, 9, 0, 9)
	knob.Position = toggleStates[stateKey] and UDim2.new(1, -11, 0.5, -4) or UDim2.new(0, 2, 0.5, -4)
	knob.BackgroundColor3 = COLOR_TOGGLE_KNOB
	knob.BorderSizePixel = 0
	knob.Parent = btn
	local knob_Corner = Instance.new("UICorner")
	knob_Corner.CornerRadius = UDim.new(1, 0)
	knob_Corner.Parent = knob
	local knob_Stroke = Instance.new("UIStroke")
	knob_Stroke.Color = COLOR_WHITE
	knob_Stroke.Thickness = 0.5
	knob_Stroke.Parent = knob
	-- Click
	btn.MouseButton1Click:Connect(function()
		toggleStates[stateKey] = not toggleStates[stateKey]
		local on = toggleStates[stateKey]
		TweenService:Create(btn, TweenInfo.new(0.15), {
			BackgroundColor3 = on and COLOR_BLUE or Color3.new(0.15, 0.17, 0.25)
		}):Play()
		TweenService:Create(knob, TweenInfo.new(0.15), {
			Position = on and UDim2.new(1, -11, 0.5, -4) or UDim2.new(0, 2, 0.5, -4)
		}):Play()
	end)
	return btn
end
local function createSlider(parent, labelText, yPos, stateKey, minVal, maxVal, isPercent)
	local container = Instance.new("Frame")
	container.Size = UDim2.new(0.85, 0, 0, 28)
	container.Position = UDim2.new(0.075, 0, 0, yPos)
	container.BackgroundTransparency = 1
	container.Parent = parent
	-- Label (oben links)
	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(1, 0, 0, 9)
	lbl.Position = UDim2.new(0, 0, 0, 0)
	lbl.BackgroundTransparency = 1
	lbl.Text = labelText
	lbl.TextColor3 = COLOR_TEXT_DIM
	lbl.Font = Enum.Font.GothamBold
	lbl.TextSize = 7
	lbl.TextXAlignment = Enum.TextXAlignment.Left
	lbl.Parent = container
	-- Wert (oben rechts)
	local valLbl = Instance.new("TextLabel")
	valLbl.Size = UDim2.new(1, 0, 0, 9)
	valLbl.Position = UDim2.new(0, 0, 0, 9)
	valLbl.BackgroundTransparency = 1
	valLbl.TextColor3 = COLOR_TEXT
	valLbl.Font = Enum.Font.Gotham
	valLbl.TextSize = 7
	valLbl.TextXAlignment = Enum.TextXAlignment.Right
	valLbl.Parent = container
	local function fmtVal(v)
		if isPercent then
			return string.format("%.1f%%", v)
		else
			return tostring(math.floor(v))
		end
	end
	valLbl.Text = fmtVal(sliderValues[stateKey])
	-- Track BG
	local track = Instance.new("Frame")
	track.Size = UDim2.new(1, 0, 0, 4)
	track.Position = UDim2.new(0, 0, 0, 20)
	track.BackgroundColor3 = COLOR_SLIDER_BG
	track.BorderSizePixel = 0
	track.Parent = container
	local track_Corner = Instance.new("UICorner")
	track_Corner.CornerRadius = UDim.new(1, 0)
	track_Corner.Parent = track
	-- Fill
	local fill = Instance.new("Frame")
	local initPct = (sliderValues[stateKey] - minVal) / (maxVal - minVal)
	fill.Size = UDim2.new(initPct, 0, 1, 0)
	fill.BackgroundColor3 = COLOR_BLUE
	fill.BorderSizePixel = 0
	fill.Parent = track
	local fill_Corner = Instance.new("UICorner")
	fill_Corner.CornerRadius = UDim.new(1, 0)
	fill_Corner.Parent = fill
	-- Thumb
	local thumb = Instance.new("Frame")
	thumb.Size = UDim2.new(0, 8, 0, 8)
	thumb.Position = UDim2.new(initPct, -4, 0.5, -4)
	thumb.BackgroundColor3 = COLOR_WHITE
	thumb.BorderSizePixel = 0
	thumb.Parent = track
	local thumb_Corner = Instance.new("UICorner")
	thumb_Corner.CornerRadius = UDim.new(1, 0)
	thumb_Corner.Parent = thumb
	-- Drag-Logik
	local dragging = false
	thumb.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or
		   input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if not dragging then return end
		if input.UserInputType == Enum.UserInputType.MouseMovement or
		   input.UserInputType == Enum.UserInputType.Touch then
			local trackPos  = track.AbsolutePosition.X
			local trackSize = track.AbsoluteSize.X
			local mouseX    = input.Position.X
			local pct = math.clamp((mouseX - trackPos) / trackSize, 0, 1)
			local val = minVal + pct * (maxVal - minVal)
			sliderValues[stateKey] = val
			fill.Size = UDim2.new(pct, 0, 1, 0)
			thumb.Position = UDim2.new(pct, -4, 0.5, -4)
			valLbl.Text = fmtVal(val)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or
		   input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	return container
end

local listeningFor = nil  -- "AlignUp" or "AlignDown"
local function startListening(keyName, btn)
	listeningFor = keyName
	btn.Text = "[...]"
	btn.TextColor3 = COLOR_BLUE
end
KeyUpBtn.MouseButton1Click:Connect(function()
	startListening("AlignUp", KeyUpBtn)
end)
KeyDownBtn.MouseButton1Click:Connect(function()
	startListening("AlignDown", KeyDownBtn)
end)
UserInputService.InputBegan:Connect(function(input, gpe)
	if listeningFor and not gpe then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			keybinds[listeningFor] = input.KeyCode
			local label = "[" .. input.KeyCode.Name .. "]"
			if listeningFor == "AlignUp" then
				KeyUpBtn.Text = label
				KeyUpBtn.TextColor3 = COLOR_TEXT
			else
				KeyDownBtn.Text = label
				KeyDownBtn.TextColor3 = COLOR_TEXT
			end
			listeningFor = nil
		end
	end
end)
local minimized = false
MinBtn.MouseButton1Click:Connect(function()
	minimized = not minimized
	if minimized then
		-- Nur TitleBar sichtbar
		TweenService:Create(MainFrame, TweenInfo.new(0.2), {
			Size = UDim2.new(0, 220, 0, 50)
		}):Play()
		MinBtn.Text = "+"
		SettingsFrame.Visible = false
	else
		TweenService:Create(MainFrame, TweenInfo.new(0.2), {
			Size = UDim2.new(0, 220, 0, 240)
		}):Play()
		MinBtn.Text = "-"
	end
end)
local settingsOpen = false
ExpandBtn.MouseButton1Click:Connect(function()
	settingsOpen = not settingsOpen
	SettingsFrame.Visible = settingsOpen and not minimized
	ExpandBtn.Text = settingsOpen and "-" or "+"
end)
